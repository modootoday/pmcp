One example from this skill

This is the skill's own text, not a summary of it. It runs: that is what verifying means here. 

import assert from 'node:assert/strict';
import Response from 'responselike';

const response = new Response({
  statusCode: 200,
  headers: {
    'Content-Type': 'text/plain',
    'X-Request-Id': 'abc123'
  },
  body: Buffer.from('Hi!'),
  url: 'https://example.com'
});

assert.equal(response.statusCode, 200);
assert.deepEqual(response.headers, {
  'content-type': 'text/plain',
  'x-request-id': 'abc123'
});
assert.deepEqual(response.body, Buffer.from('Hi!'));
assert.equal(response.url, 'https://example.com');
