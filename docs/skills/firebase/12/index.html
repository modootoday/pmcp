One example from this skill

This is the skill's own text, not a summary of it. It runs: that is what verifying means here. 

import assert from 'node:assert/strict';
import { initializeApp, getApp, getApps, deleteApp } from 'firebase/app';

const app = initializeApp({
  apiKey: 'test-api-key',
  authDomain: 'example.firebaseapp.com',
  projectId: 'example-project'
}, 'skill-example');

assert.equal(getApp('skill-example'), app);
assert.equal(getApps().includes(app), true);

await deleteApp(app);
assert.equal(getApps().some(candidate => candidate === app), false);
