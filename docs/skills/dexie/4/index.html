One example from this skill

This is the skill's own text, not a summary of it. It runs: that is what verifying means here. 

import assert from "node:assert/strict";
import DexieDefault, { Dexie } from "dexie";

assert.equal(DexieDefault, Dexie);

const db = new Dexie("DexieSkillCore");
db.version(1).stores({
  friends: "++id, name, age",
  pets: "++id, name, kind",
});

const friends = db.table("friends");
assert.equal(friends.name, "friends");
assert.deepEqual(
  db.tables.map((table) => table.name).sort(),
  ["friends", "pets"],
);
